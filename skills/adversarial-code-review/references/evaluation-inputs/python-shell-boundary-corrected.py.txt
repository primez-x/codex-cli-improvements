from __future__ import annotations

import re
import subprocess
from pathlib import Path


REVISION = re.compile(r"[A-Za-z0-9][A-Za-z0-9._/-]{0,127}\Z")
COMMIT = re.compile(r"[0-9a-f]{40}\Z")


def inspect_revision(revision: str, repository: str | Path = ".") -> str:
    """Resolve one inert revision argument and show its exact commit identity."""
    if not isinstance(revision, str) or REVISION.fullmatch(revision) is None:
        raise ValueError("revision contains unsupported characters")
    resolved = subprocess.run(
        ["git", "rev-parse", "--verify", f"{revision}^{{commit}}"],
        cwd=repository,
        check=False,
        text=True,
        capture_output=True,
    )
    commit = resolved.stdout.strip()
    if resolved.returncode != 0 or COMMIT.fullmatch(commit) is None:
        raise ValueError("revision does not resolve to a commit")
    shown = subprocess.run(
        ["git", "show", "--no-patch", "--format=%H", commit, "--"],
        cwd=repository,
        check=True,
        text=True,
        capture_output=True,
    )
    return shown.stdout
